var CASES = [
  { el: document.getElementById('case-calyx'), anchors: [
      ['top', 0], ['#studio', 0], ['#studio + section', 0], ['#index', 0], ['#commission', 0] ] },
  { el: document.getElementById('case-horai'), anchors: [

/* Andra talet: förskjutning i ramhöjder. 0,30 = husets stycke står en tredjedel
   ned så ordljuset spelar; -0,40 = citatets eget "formade läge" (horai.html rad

function measureCase(C){
  var f = C.el.querySelector('iframe'), w = f && f.contentWindow, d = f && f.contentDocument;
  if (!d || !d.documentElement) return;
  var ih = w.innerHeight, max = d.documentElement.scrollHeight - ih;
  C.f = f; C.w = w; C.y = C.anchors.map(function(a){
    var t = a[0] === 'top' ? 0 : (d.querySelector(a[0]) || {}).offsetTop || 0;
    return Math.max(0, Math.min(max, Math.round(t + a[1] * ih)));
  });
}
function caseY(C, p){
  var n = C.y.length, k = Math.min(n - 1, Math.floor(p * n)), t = p * n - k;
  if (k === 0) return C.y[0];
  var u = Math.min(1, t / 0.45); u = u * u * (3 - 2 * u);   /* resa 45 %, stå 55 % */
  return C.y[k - 1] + (C.y[k] - C.y[k - 1]) * u;
}

## Repo conventions

- Motorn `scroll/scrollcraft.js`, `scrollcraft.css`, `scroll/sc-smooth.js` och `sites/*`
  orörda. Inga nya `@property`.

# V3d · PARITET

Tabellen granskaren (Astra) och lead dömer mot. En rad per skärm i V3d. Varje rad har en
referensbild ur `runs/V3c/shots/fasA/` (alla sökvägar nedan är relativa till den mappen) och
ett uttalat drag som V3d ska matcha. **Referensen visar nivån, inte formen**: vi lånar hur den
iscensätter skärmen (skala, vad som äger viewporten, hur mycket luft), aldrig dess palett,
typsnitt, knappar eller copy.

## Hur en rad döms

1. Byggaren skjuter vår skärm vid 1440x900 (och 390x844 där kolumnen 390 är ifylld) i det läge
   raden anger, och lägger den **bredvid** referensbilden i samma höjd med PIL:
   `shots/r1/par/<rad-id>-<bredd>.png` (vänster V3d, höger referensen, 24 px mellanrum,
   radens id och referensens filnamn i bildens underkant i 14 px).
2. Den som dömer läser par-bilden som bild och skriver en rad: **håller nivån** eller
   **håller inte, för att …** med det mätbara skälet (en storlek, en andel av viewporten, ett
   element som saknas).
3. **Slutvillkor:** varje rad i tabellen kan läsas "håller nivån" i sin par-bild. En rad som
   håller i 1440 men inte i 390 håller inte.

- **Hela sidan:** hjul i steg om 150 px vid 1440 och 390 (`whole-<w>/`, `sheet-whole-*.png`). Rubrikkollisioner kontrollerades med `elementFromPoint`, och horisontell scroll mättes.
